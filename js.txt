import React, {
  useEffect,
  useState,
  createContext,
  useContext,
  lazy,
  Suspense,
} from "react";

import {
  BrowserRouter,
  Routes,
  Route,
  Link,
  useParams,
} from "react-router-dom";

/* =========================
   CONTEXT
========================= */

const CartContext = createContext();

function CartProvider({ children }) {
  const [cart, setCart] = useState([]);

  function addToCart(product) {
    setCart((prev) => [...prev, product]);
  }

  return (
    <CartContext.Provider value={{ cart, addToCart }}>
      {children}
    </CartContext.Provider>
  );
}

function useCart() {
  return useContext(CartContext);
}

/* =========================
   API
========================= */

const BASE_URL = "https://fakestoreapi.com";

async function getProducts() {
  const res = await fetch(`${BASE_URL}/products`);
  return res.json();
}

async function getProduct(id) {
  const res = await fetch(`${BASE_URL}/products/${id}`);
  return res.json();
}

/* =========================
   COMPONENTS
========================= */

function Navbar() {
  const { cart } = useCart();

  return (
    <nav className="bg-black text-white px-6 py-4 flex justify-between items-center">
      <Link to="/" className="text-2xl font-bold">
        MyStore
      </Link>

      <div className="flex gap-6">
        <Link to="/">Home</Link>
        <Link to="/products">Products</Link>
        <Link to="/cart">Cart ({cart.length})</Link>
      </div>
    </nav>
  );
}

function ProductCard({ product }) {
  return (
    <div className="border rounded-xl p-4 shadow hover:shadow-xl transition duration-300 bg-white">
      <img
        src={product.image}
        alt={product.title}
        loading="lazy"
        className="h-48 mx-auto object-contain"
      />

      <h2 className="font-bold mt-4 line-clamp-2">
        {product.title}
      </h2>

      <p className="text-lg font-semibold mt-2">
        ${product.price}
      </p>

      <Link
        to={`/products/${product.id}`}
        className="inline-block mt-4 bg-black text-white px-4 py-2 rounded-lg"
      >
        View Details
      </Link>
    </div>
  );
}

/* =========================
   PAGES
========================= */

function Home() {
  return (
    <div className="min-h-screen flex flex-col justify-center items-center bg-gray-100 text-center p-6">
      <h1 className="text-5xl font-bold mb-6">
        E-Commerce Product Catalog
      </h1>

      <p className="text-lg text-gray-700 max-w-2xl">
        Modern React E-Commerce frontend with routing,
        cart management, lazy loading, responsive design,
        and deployment-ready optimization.
      </p>

      <Link
        to="/products"
        className="mt-8 bg-black text-white px-8 py-4 rounded-xl text-lg"
      >
        Explore Products
      </Link>
    </div>
  );
}

function Products() {
  const [products, setProducts] = useState([]);
  const [search, setSearch] = useState("");

  useEffect(() => {
    getProducts().then(setProducts);
  }, []);

  const filteredProducts = products.filter((product) =>
    product.title.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <div className="min-h-screen bg-gray-100 p-6">
      <h1 className="text-4xl font-bold mb-6">
        Products
      </h1>

      <input
        type="text"
        placeholder="Search products..."
        className="w-full md:w-96 p-3 border rounded-lg mb-8"
        value={search}
        onChange={(e) => setSearch(e.target.value)}
      />

      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-6">
        {filteredProducts.map((product) => (
          <ProductCard
            key={product.id}
            product={product}
          />
        ))}
      </div>
    </div>
  );
}

function ProductDetails() {
  const { id } = useParams();
  const [product, setProduct] = useState(null);

  const { addToCart } = useCart();

  useEffect(() => {
    getProduct(id).then(setProduct);
  }, [id]);

  if (!product) {
    return (
      <div className="p-10 text-center text-2xl">
        Loading...
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-white p-8">
      <div className="max-w-6xl mx-auto grid md:grid-cols-2 gap-12">
        <div className="bg-gray-100 rounded-2xl p-8">
          <img
            src={product.image}
            alt={product.title}
            className="h-96 mx-auto object-contain"
          />
        </div>

        <div>
          <h1 className="text-4xl font-bold">
            {product.title}
          </h1>

          <p className="text-gray-600 mt-6 leading-7">
            {product.description}
          </p>

          <p className="text-3xl font-bold mt-8">
            ${product.price}
          </p>

          <button
            onClick={() => addToCart(product)}
            className="mt-8 bg-black text-white px-8 py-4 rounded-xl hover:opacity-90"
          >
            Add To Cart
          </button>
        </div>
      </div>
    </div>
  );
}

function Cart() {
  const { cart } = useCart();

  const total = cart
    .reduce((sum, item) => sum + item.price, 0)
    .toFixed(2);

  return (
    <div className="min-h-screen bg-gray-100 p-6">
      <h1 className="text-4xl font-bold mb-8">
        Shopping Cart
      </h1>

      {cart.length === 0 ? (
        <p className="text-xl">Your cart is empty.</p>
      ) : (
        <>
          <div className="grid gap-4">
            {cart.map((item, index) => (
              <div
                key={index}
                className="bg-white p-4 rounded-xl shadow flex gap-4 items-center"
              >
                <img
                  src={item.image}
                  alt={item.title}
                  className="h-20 w-20 object-contain"
                />

                <div>
                  <h2 className="font-bold">
                    {item.title}
                  </h2>

                  <p className="text-lg font-semibold mt-2">
                    ${item.price}
                  </p>
                </div>
              </div>
            ))}
          </div>

          <div className="mt-10 text-2xl font-bold">
            Total: ${total}
          </div>
        </>
      )}
    </div>
  );
}

function NotFound() {
  return (
    <div className="min-h-screen flex justify-center items-center text-center">
      <div>
        <h1 className="text-6xl font-bold">
          404
        </h1>

        <p className="text-xl mt-4">
          Page not found
        </p>

        <Link
          to="/"
          className="inline-block mt-6 bg-black text-white px-6 py-3 rounded-lg"
        >
          Go Home
        </Link>
      </div>
    </div>
  );
}

/* =========================
   LAZY LOADING
========================= */

const LazyProducts = lazy(() =>
  Promise.resolve({ default: Products })
);

const LazyProductDetails = lazy(() =>
  Promise.resolve({ default: ProductDetails })
);

/* =========================
   APP
========================= */

export default function App() {
  return (
    <CartProvider>
      <BrowserRouter>
        <Navbar />

        <Suspense
          fallback={
            <div className="p-10 text-center text-2xl">
              Loading...
            </div>
          }
        >
          <Routes>
            <Route path="/" element={<Home />} />

            <Route
              path="/products"
              element={<LazyProducts />}
            />

            <Route
              path="/products/:id"
              element={<LazyProductDetails />}
            />

            <Route path="/cart" element={<Cart />} />

            <Route path="*" element={<NotFound />} />
          </Routes>
        </Suspense>
      </BrowserRouter>
    </CartProvider>
  );
}
